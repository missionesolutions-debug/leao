# Testes para usuários
def test_create_user():
    from app.services.user_service import UserService
    class DummyRepo:
        def create(self, data): return {"id": 1, "email": data.email}
    service = UserService(DummyRepo())
    class DummyUser: email = "test@example.com"
    user = service.create_user(DummyUser())
    assert user["id"] == 1
    assert user["email"] == "test@example.com"

def test_get_user():
    from app.services.user_service import UserService
    class DummyRepo:
        def get_user_by_id(self, user_id): return {"id": user_id, "email": "test@example.com"}
    service = UserService(DummyRepo())
    user = service.get_user(1)
    assert user["id"] == 1

def test_update_user():
    from app.services.user_service import UserService
    class DummyRepo:
        def update_user(self, user_id, data): return {"id": user_id, "email": data.email}
    service = UserService(DummyRepo())
    class DummyUser: email = "updated@example.com"
    user = service.update_user(1, DummyUser())
    assert user["email"] == "updated@example.com"

def test_delete_user():
    from app.services.user_service import UserService
    class DummyRepo:
        def delete_user(self, user_id): return True
    service = UserService(DummyRepo())
    result = service.delete_user(1)
    assert result is True
